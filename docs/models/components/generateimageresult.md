```typescript
import { GenerateImageResult } from "@cloudinary/asset-management/models/components";

let value: GenerateImageResult = {
  data: {
    assets: [],
  },
  limits: {
    addonsQuota: [
      {
        type: "image_generation",
        usedByRequest: 1,

        limit: 50,
      },
    ],
  },
  notices: [

      severity: "warning",
      text:

};
```
