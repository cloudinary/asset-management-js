# ContractCreatePublicPageRequestPayload

Settings for the public page. Every field is optional; omitted fields use their documented default.


## Example Usage

```typescript
import { ContractCreatePublicPageRequestPayload } from "@cloudinary/asset-management/models/components";

let value: ContractCreatePublicPageRequestPayload = {
  name: "Summer Campaign Preview",
  availableFrom: new Date("2026-03-20T00:00:00Z"),
  expiresAt: new Date("2026-04-20T00:00:00Z"),
  showDownloadOriginalOption: true,
  downloadTransformations: [
    {
      name: "thumbnail_400",
    },
    {
      name: "trimmed_preview",
      types: [
        "video",
      ],
    },
  ],
  createDownloadTransformations: true,
};
```
