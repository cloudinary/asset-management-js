# GenerateImageRequest

Parameters for an image-generation request. Only `prompt` is required;
all other fields fall back to documented defaults.


## Example Usage

```typescript
import { GenerateImageRequest } from "@cloudinary/asset-management/models/components";

let value: GenerateImageRequest = {
  prompt: "A photorealistic sunset over a mountain lake, 8K detail",
  model: {
    family: "flux",
    tier: "premium",
  },
  imageSize: {
    aspectRatio: "16:9",
    resolution: "2K",
  },
  format: "png",
  target: {
    targetType: "temporary",
  },
  seed: 42,
  notificationUrl: "https://path.to/webhook",
};
```
