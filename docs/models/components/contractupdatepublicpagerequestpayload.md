# ContractUpdatePublicPageRequestPayload

Partial-update payload. Only the fields you include are forwarded and changed; any field you omit is left unchanged. Sending null does not clear a value.


## Example Usage

```typescript
import { ContractUpdatePublicPageRequestPayload } from "@cloudinary/asset-management/models/components";

let value: ContractUpdatePublicPageRequestPayload = {
  name: "Summer Campaign Preview",
  availableFrom: new Date("2026-03-20T00:00:00Z"),
  expiresAt: new Date("2026-04-20T00:00:00Z"),
  showDownloadOriginalOption: true,
  downloadTransformations: [
    {
      name: "thumbnail_400",
    },
    {
      name: "trimmed_preview",
      types: [
        "video",
      ],
    },
  ],
  createDownloadTransformations: true,
  protectionMode: "email_otp",
  allowedEmails: [
    "alice@example.com",
    "bob@example.com",
  ],
  allowedDomains: [
    "example.com",
  ],
  allowAny: false,
  customLayoutUrl:
    "https://res.cloudinary.com/demo/raw/upload/layouts/summer.html",
};
```
