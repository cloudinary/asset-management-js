# ImageToImageRequest

Parameters for an image-to-image (reference-guided) request. Both
`prompt` (the edit instruction) and `reference_images` are required; all
other fields fall back to the same defaults as `text_to_image`.


## Example Usage

```typescript
import { ImageToImageRequest } from "@cloudinary/asset-management/models/components";

let value: ImageToImageRequest = {
  prompt:
    "Place the product from [1] on a marble kitchen counter, soft morning light",
  referenceImages: [
    {
      sourceType: "url",
      url: "https://example.com/product.png",
    },
  ],
  model: {
    family: "flux",
    tier: "premium",
  },
  imageSize: {
    width: 1280,
    height: 720,
  },
  format: "png",
  target: {
    targetType: "managed_asset",
  },
  seed: 42,
  notificationUrl: "https://path.to/webhook",
};
```
